cmake_minimum_required(VERSION 3.16)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.2" CACHE STRING "Minimum macOS version")
endif()
project(DriftEQ VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(drift_core native/core/Engine.cpp)
target_include_directories(drift_core PUBLIC native/core)
if(MSVC)
  target_compile_options(drift_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(drift_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(drift_tests native/tests/Tests.cpp)
  target_link_libraries(drift_tests PRIVATE drift_core Threads::Threads)
  add_test(NAME drift_core COMMAND drift_tests)
endif()
if(WIN32)
  add_executable(DriftEQ WIN32 native/windows/Main.cpp native/windows/Wasapi.cpp)
  target_compile_definitions(DriftEQ PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(DriftEQ PRIVATE drift_core ole32 uuid avrt comctl32 shell32 propsys)
  target_compile_options(DriftEQ PRIVATE /W4 /permissive- /utf-8 /wd4324)
  # The APO DLL uses the dynamic CRT and no embedded manifest, as required by
  # Windows audio effect deployment. Keep the portable cable app independent.
  add_library(DriftApo SHARED native/windows/apo/DriftApo.cpp native/windows/apo/DriftApo.def
    native/windows/apo/Settings.cpp native/core/Engine.cpp)
  target_include_directories(DriftApo PRIVATE native/core)
  target_compile_definitions(DriftApo PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(DriftApo PRIVATE /W4 /permissive- /utf-8)
  target_link_options(DriftApo PRIVATE /MANIFEST:NO)
  target_link_libraries(DriftApo PRIVATE ole32 uuid advapi32 AudioBaseProcessingObjectV140 audioeng)
  set_property(TARGET DriftApo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  add_executable(DriftControl WIN32 native/windows/apo/Control.cpp native/windows/apo/Settings.cpp)
  target_compile_definitions(DriftControl PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(DriftControl PRIVATE /W4 /permissive- /utf-8)
  target_link_libraries(DriftControl PRIVATE drift_core ole32 uuid advapi32 shell32 comctl32)
  if(BUILD_TESTING)
    add_executable(drift_apo_tests native/windows/apo/Tests.cpp)
    target_compile_definitions(drift_apo_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(drift_apo_tests PRIVATE native/windows/apo)
    target_link_libraries(drift_apo_tests PRIVATE drift_core ole32 uuid propsys)
    add_dependencies(drift_apo_tests DriftApo)
    add_test(NAME drift_apo COMMAND drift_apo_tests $<TARGET_FILE:DriftApo>)
    add_test(NAME drift_apo_setup COMMAND powershell.exe -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/native/windows/apo/Setup.Tests.ps1")
  endif()
  install(TARGETS DriftEQ DriftApo DriftControl RUNTIME DESTINATION .)
elseif(APPLE)
  enable_language(OBJCXX)
  add_executable(DriftEQ MACOSX_BUNDLE native/macos/Main.mm native/macos/TapAudio.mm)
  target_compile_options(DriftEQ PRIVATE -fobjc-arc -Wall -Wextra)
  target_link_libraries(DriftEQ PRIVATE drift_core "-framework Cocoa" "-framework CoreAudio" "-framework AudioToolbox")
  set_target_properties(DriftEQ PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/native/macos/Info.plist" MACOSX_BUNDLE_GUI_IDENTIFIER "com.nosam1998.drifteq" MACOSX_BUNDLE_BUNDLE_NAME "Drift EQ")
  install(TARGETS DriftEQ BUNDLE DESTINATION .)
endif()
